## Troubleshooting

### Error: `429 Too many requests`

When you exceed the per-second rate limit, Web Search responds with HTTP status code `429 Too many requests`. Rate limits are based on your [Workers plan](/workers/platform/pricing/).

The example below shows how to handle rate limiting gracefully by reading the `Retry-After` header and retrying the request after the indicated delay. If the header is missing, the example falls back to exponential backoff.

<Tabs syncKey="integration">
<TabItem label="Workers Binding">

```ts
async function searchWithRetry(env: Env, query: string, attempts = 3) {
	for (let i = 0; i < attempts; i++) {
		try {
			return await env.WEBSEARCH.search({ query, limit: 5 });
		} catch (error: any) {
			if (error.status !== 429 || i === attempts - 1) throw error;
			const retryAfter =
				Number(error.headers?.get("Retry-After")) || 2 ** i;
			console.log(
				`Rate limited. Waiting ${retryAfter}s before retry ${i + 1}.`,
			);
			await new Promise((resolve) =>
				setTimeout(resolve, retryAfter * 1000),
			);
		}
	}
}
```

```ts
async function searchWithRetry(query: string, attempts = 3) {
	for (let i = 0; i < attempts; i++) {
		const response = await fetch(
			"https://api.cloudflare.com/client/v4/accounts/<accountId>/websearch/search",
			{
				method: "POST",
				headers: {
					"Content-Type": "application/json",
					Authorization: "Bearer <your-token>",
				},
				body: JSON.stringify({ query, limit: 5 }),
			},
		);

		if (response.status !== 429) return response;
		if (i === attempts - 1) return response;

		const retryAfter =
			Number(response.headers.get("Retry-After")) || 2 ** i;
		console.log(
			`Rate limited. Waiting ${retryAfter}s before retry ${i + 1}.`,
		);
		await new Promise((resolve) => setTimeout(resolve, retryAfter * 1000));
	}
}
```
